import { useSearchParams, useNavigate } from 'react-router-dom';
import axios from 'axios';

const VerifyDonation = () => {
  const [searchParams] = useSearchParams();
  const [status, setStatus] = useState('Verifying your payment, please wait...');
  const navigate = useNavigate();

  // Paystack appends ?reference=xxxx or ?trxref=xxxx to the callback URL
  const reference = searchParams.get('reference') || searchParams.get('trxref') || localStorage.getItem('pending_donation_ref');

  useEffect(() => {
    const verifyPayment = async () => {
      if (!reference) {
        setStatus('No payment reference found.');
        return;
      }

      try {
        // Call your backend verify controller: GET /api/donations/verify/:reference
        const res = await axios.get(`http://localhost:5000/api/donations/verify/${reference}`);

        if (res.status === 200) {
          setStatus('Payment Successful! Thank you for your donation.');
          localStorage.removeItem('pending_donation_ref');
          
          // Optional: redirect home after 3 seconds
          setTimeout(() => navigate('/'), 3000);
        }
      } catch (err) {
        console.error('Verification error:', err);
        setStatus('Payment verification failed or was canceled.');
      }
    };

    verifyPayment();
  }, [reference, navigate]);

  return (
    <div style={{ textAlign: 'center', padding: '50px' }}>
      <h2>{status}</h2>
    </div>
  );
};
